cmake_minimum_required(VERSION 3.22)
project(iimt_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
include(cpp/control.cmake)
include(cpp/model.cmake)
include(cpp/viewer.cmake)
# Only native binaries for the wheel's platform; never load the other vendor build.
if(WIN32)
  install(FILES vendor/windows/iimtcl.dll DESTINATION iimt/vendor)
else()
  install(FILES vendor/linux/libiimtcl.so DESTINATION iimt/vendor)
endif()
